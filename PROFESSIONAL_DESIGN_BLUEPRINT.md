### **2. Admin Address in Footer**
```javascript
// New Supabase table: settings
// Add fields: address_line1, address_line2, city, state, pincode

// Fetch in app.js loadLiveData()
async function loadLiveData(onReady) {
  // ... existing code ...
  
  const setRes = await client.from("settings").select("*");
  if (setRes.data) {
    const settings = {};
    setRes.data.forEach(row => { settings[row.key] = row.value; });
    
    DEMO_CONTACT.address = {
      line1: settings.address_line1 || "",
      line2: settings.address_line2 || "",
      city: settings.city || "",
      state: settings.state || "",
      pincode: settings.pincode || ""
    };
  }
}

// Render in footer
function buildFooter() {
  const addr = DEMO_CONTACT.address;
  const addressHTML = addr ? `
    <div style="display:flex;gap:8px;align-items:flex-start;">
      <span style="color:var(--saffron);">📍</span>
      <div>
        ${addr.line1}<br/>
        ${addr.line2 ? addr.line2 + '<br/>' : ''}
        ${addr.city}, ${addr.state} - ${addr.pincode}
      </div>
    </div>
  ` : '';
  
  // ... rest of footer with addressHTML ...
}
```
